call_priority Formula

eligible = contacts.filter(c =>
  c.active_status === "Active" &&
  c.union_eligible === true &&
  c.do_not_call === false
)
eligible.sort((a, b) => b.seniority_years - a.seniority_years)
eligible.forEach((c, i) => c.call_priority = i + 1)
nonEligible.forEach(c => c.call_priority = 9999)
